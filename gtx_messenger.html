Complete Realtime Database Security Rules (`database.rules.json`) Open your Firebase Console → Realtime Database → Rules tab and paste the JSON below. It enforces strict security across users, identities, usernameIndex, conversations, conversationMembers, and messages—ensuring only conversation participants can read messages and users can never send messages using identities they do not own.
database.rules.json 
{
  "rules": {
    ".read": false,
    ".write": false,

    "users": {
      "$uid": {
        ".read": "auth != null && auth.uid === $uid",
        ".write": "auth != null && auth.uid === $uid"
      }
    },

    "identities": {
      ".indexOn": ["ownerUid", "username"],
      ".read": "auth != null",
      "$identityId": {
        ".read": "auth != null && (data.child('ownerUid').val() === auth.uid || data.child('isPublic').val() === true)",
        ".write": "auth != null && ((!data.exists() && newData.child('ownerUid').val() === auth.uid) || (data.exists() && data.child('ownerUid').val() === auth.uid && (!newData.exists() || newData.child('ownerUid').val() === auth.uid)))"
      }
    },

    "usernameIndex": {
      ".read": "auth != null",
      "$username": {
        ".write": "auth != null && ((!data.exists() && newData.child('ownerUid').val() === auth.uid) || (data.exists() && data.child('ownerUid').val() === auth.uid && (!newData.exists() || newData.child('ownerUid').val() === auth.uid)))"
      }
    },

    "conversationMembers": {
      "$conversationId": {
        ".read": "auth != null && data.child(auth.uid).val() === true",
        ".write": "auth != null && ((!data.exists() && newData.child(auth.uid).val() === true) || data.child(auth.uid).val() === true)"
      }
    },

    "conversations": {
      ".read": "auth != null",
      "$conversationId": {
        ".read": "auth != null && root.child('conversationMembers').child($conversationId).child(auth.uid).val() === true",
        ".write": "auth != null && (root.child('conversationMembers').child($conversationId).child(auth.uid).val() === true || newData.parent().parent().child('conversationMembers').child($conversationId).child(auth.uid).val() === true)"
      }
    },

    "messages": {
      "$conversationId": {
        ".read": "auth != null && root.child('conversationMembers').child($conversationId).child(auth.uid).val() === true",
        "$messageId": {
          ".write": "auth != null && root.child('conversationMembers').child($conversationId).child(auth.uid).val() === true && ((!data.exists() && newData.child('senderUid').val() === auth.uid && root.child('identities').child(newData.child('senderIdentityId').val()).child('ownerUid').val() === auth.uid) || (data.exists() && data.child('senderUid').val() === auth.uid && newData.child('senderUid').val() === auth.uid && newData.child('senderIdentityId').val() === data.child('senderIdentityId').val()))"
        }
      }
    }
  }
}

Exact Realtime Database Tree Architecture (`gtxmessenger-default-rtdb`)

GTX-MESSENGER (https://gtxmessenger-default-rtdb.firebaseio.com)
├── users/$userId
│    ├── email, createdAt, defaultIdentityId
├── identities/$identityId
│    ├── identityId, ownerUid, username, displayName, photoURL, bio, createdAt, isPublic
├── usernameIndex/$username (lowercase unique key, e.g. "alex123")
│    ├── identityId, ownerUid, displayName, photoURL, bio
├── conversations/$conversationId
│    ├── conversationId, createdAt, updatedAt, lastMessage, lastMessageAt, participantIdentities
├── conversationMembers/$conversationId
│    └── $userId: true
└── messages/$conversationId/$messageId
     └── messageId, conversationId, senderUid, senderIdentityId, senderUsername, senderDisplayName, senderPhotoURL, text, createdAt, deleted

Two-Account Live Search & Security Verification Guide

  1. Account A: Register in your main browser window (e.g. create persona @alex123 and a second persona @shadow_ops). Both are atomically written to /identities and /usernameIndex.
  2. Account B: Open an Incognito/Private window and register a second account (e.g. persona @john_gtx).
  3. Real-Time Prefix Search: In Account B, click Directory and type alex. The live onValue listener on usernameIndex immediately streams @alex123 with avatar and display name.
  4. Profile & Start Chat: Click on @alex123 to open their profile card, choose which of your personas to speak as, and click Start Chat.
  5. Rename / Delete Cleanup: Rename @alex123 to @alex_pro in Multi-Identity Studio—the old usernameIndex/alex123 node is atomically set to null and replaced by usernameIndex/alex_pro.
